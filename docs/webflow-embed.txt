<script>
(function () {
  function initializeForm() {
    const API_URL = "https://mellowart-dev.mellowartmarket.workers.dev/api/submit";
    const CLIENT_KEY = "PASTE_YOUR_EXISTING_CLIENT_KEY_HERE";
    const form = document.getElementById("wf-form-artist-application");
    if (!form || form.dataset.dashboardSubmitBound === "true") return;
    form.dataset.dashboardSubmitBound = "true";
    let submitting = false;

    form.addEventListener("submit", async function (e) {
      e.preventDefault();
      e.stopImmediatePropagation();
      if (submitting || !form.reportValidity()) return;
      const submitBtn = form.querySelector('input[type="submit"], button[type="submit"]');
      const originalLabel = submitBtn
        ? submitBtn.tagName === "INPUT" ? submitBtn.value : submitBtn.textContent : "";
      function setButtonLabel(label) {
        if (!submitBtn) return;
        if (submitBtn.tagName === "INPUT") submitBtn.value = label;
        else submitBtn.textContent = label;
      }
      submitting = true;
      setButtonLabel("Submitting...");
      if (submitBtn) submitBtn.disabled = true;
      try {
        const buddySection = form.querySelector("#Stall-Buddy-Details");
        function getMainValue(selector) {
          for (const el of form.querySelectorAll(selector)) {
            if (!el.disabled && !buddySection?.contains(el)) return (el.value || "").trim();
          }
          return "";
        }
        function getBuddyValue(selector) {
          const el = buddySection?.querySelector(selector);
          return !el || el.disabled ? "" : (el.value || "").trim();
        }
        const emailFields = Array.from(
          form.querySelectorAll('input[type="email"]:not([disabled])')
        ).filter(function (el) { return !buddySection?.contains(el); });
        const email = emailFields[0]?.value.trim() || "";
        const confirmEmail = emailFields[1]?.value.trim() || "";
        if (!email || !confirmEmail) throw new Error("Please enter and confirm your email address.");
        if (email.toLowerCase() !== confirmEmail.toLowerCase()) {
          throw new Error("Email addresses do not match. Please check and try again.");
        }
        const sharing = getMainValue('[name="sharing"]').toLowerCase();
        if (sharing !== "yes" && sharing !== "no") {
          throw new Error("Please select whether you are sharing a stall.");
        }
        let buddyEmail = "";
        let buddyConfirmEmail = "";
        if (sharing === "yes") {
          if (!buddySection) throw new Error("Stall buddy details are missing from this form.");
          const fields = buddySection.querySelectorAll('input[type="email"]:not([disabled])');
          buddyEmail = fields[0]?.value.trim() || "";
          buddyConfirmEmail = fields[1]?.value.trim() || "";
          if (!buddyEmail || !buddyConfirmEmail) {
            throw new Error("Please enter and confirm your stall buddy's email address.");
          }
          if (buddyEmail.toLowerCase() !== buddyConfirmEmail.toLowerCase()) {
            throw new Error("Stall buddy email addresses do not match.");
          }
        }
        // Bind this hidden input or text element to the CMS event slug.
        const slugInput = form.querySelector('[name="eventSlug"]');
        const eventSlug = (slugInput?.value || "").trim() ||
          (document.getElementById("eventSlug")?.textContent || "").trim();
        if (!eventSlug) throw new Error("Event information is missing. Please refresh the page.");
        const data = new FormData();
        data.set("eventSlug", eventSlug);
        const mainFields = {
          firstName: '[name="first-name"]',
          lastName: '[name="last-name"]',
          appliedBefore: '[name="first-timer"]',
          brandName: '[name="brand-name"]',
          website: '[name="website"]',
          instagram: '[name="instagram"]',
          bio: '[name="artist-bio"]',
          primaryCategory: '[name="category-01"]',
          secondaryCategory: '[name="category-02"]',
          productDescription: '[name="product-info"]',
          additionalNotes: '[name="other"]',
          secondStallPreference: '[name="second-pref"]',
          offerMiniIfUnavailable: '[name="half-table-permission"]',
          hasInsurance: '[name="insurance"]'
        };
        for (const [key, selector] of Object.entries(mainFields)) data.set(key, getMainValue(selector));
        data.set("email", email);
        data.set("confirmEmail", confirmEmail);
        data.set("sharingStall", sharing);
        const preference = getMainValue('input[name="first-pref"]:checked');
        if (!preference) throw new Error("Please select your first stall preference.");
        data.set("firstStallPreference", preference);
        const consentDebut = form.querySelector('[name="Checkbox---debut"]');
        const consents = form.querySelectorAll('[name="Checkbox---buddy-stall-size"]');
        data.set("consentDebut", consentDebut?.checked ? "true" : "false");
        data.set("consentSharing", consents[0]?.checked ? "true" : "false");
        data.set("consentSetupGuide", consents[1]?.checked ? "true" : "false");

        const MAX_FILE_BYTES = 10 * 1024 * 1024;
        function validateFile(file) {
          if (file.size === 0) throw new Error(file.name + ": the file is empty.");
          if (file.size > MAX_FILE_BYTES) throw new Error(file.name + ": maximum file size is 10 MB.");
        }
        const portfolioInput = Array.from(form.querySelectorAll(
          'input[type="file"][name="portfolio-file"]:not([disabled])'
        )).find(function (input) { return !buddySection?.contains(input); });
        const portfolio = portfolioInput?.files?.[0];
        if (!portfolio) throw new Error("Please upload a portfolio document.");
        validateFile(portfolio);
        data.set("portfolio", portfolio, portfolio.name);
        // Separate native inputs or a custom multi-file input are both supported.
        for (const input of form.querySelectorAll('input[type="file"]')) {
          if (!/^insurance-file(?:-[1-9]\d*)?$/.test(input.name)) continue;
          if (input.disabled || buddySection?.contains(input)) continue;
          for (const file of Array.from(input.files || [])) {
            validateFile(file);
            data.append("insurance", file, file.name);
          }
        }
        if (sharing === "yes") {
          const buddyFields = {
            buddyFirstName: '[name="buddy-first-name"]',
            buddyLastName: '[name="buddy-last-name"]',
            buddyAppliedBefore: '[name="buddy-first-timer"]',
            buddyBrandName: '[name="buddy-brand-name"]',
            buddyWebsite: '[name="buddy-website"]',
            buddyInstagram: '[name="buddy-instagram"]',
            buddyBio: '[name="buddy-artist-bio"]',
            buddyPrimaryCategory: '[name="buddy-category-01"]',
            buddySecondaryCategory: '[name="buddy-category-02"]',
            buddyProductDescription: '[name="buddy-product-info"]'
          };
          for (const [key, selector] of Object.entries(buddyFields)) data.set(key, getBuddyValue(selector));
          data.set("buddyEmail", buddyEmail);
          data.set("buddyConfirmEmail", buddyConfirmEmail);
          const buddyPortfolioInput = buddySection.querySelector(
            'input[type="file"][name="buddy-portfolio-file"]:not([disabled])'
          );
          const buddyPortfolio = buddyPortfolioInput?.files?.[0];
          if (!buddyPortfolio) throw new Error("Please upload your stall buddy's portfolio.");
          validateFile(buddyPortfolio);
          data.set("buddyPortfolio", buddyPortfolio, buddyPortfolio.name);
        }
        // Do not set Content-Type: fetch supplies the multipart boundary.
        const res = await fetch(API_URL, {
          method: "POST", headers: { "X-Client-Key": CLIENT_KEY }, body: data
        });
        const result = await res.json().catch(function () { return null; });
        if (!res.ok) {
          let message = result?.error || "Submission failed (" + res.status + ").";
          if (result?.issues?.fieldErrors) {
            const messages = Object.entries(result.issues.fieldErrors).map(function ([field, errors]) {
              return field + ": " + (Array.isArray(errors) ? errors.join(", ") : String(errors));
            });
            if (messages.length) message += "\n\n" + messages.join("\n");
          }
          throw new Error(message);
        }
        if (result?.ok !== true || !result?.id) {
          throw new Error("The server did not confirm your submission. Please contact us before submitting again.");
        }
        window.location.assign("/application-success");
      } catch (err) {
        console.error("Application submission failed:", err);
        alert(err instanceof Error ? err.message : "Something went wrong. Please try again.");
        submitting = false;
        setButtonLabel(originalLabel);
        if (submitBtn) submitBtn.disabled = false;
      }
    }, true);
  }
  if (document.readyState === "loading") document.addEventListener("DOMContentLoaded", initializeForm);
  else initializeForm();
})();
</script>
